cmake_minimum_required(VERSION 3.5)

project(Planetary VERSION 0.1 LANGUAGES CXX)

add_custom_target(gen_version ALL
    COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/make_version.pl
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
    COMMENT "Generating version.h"
)

set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets LinguistTools Network Core)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets LinguistTools Network Core)

set(TS_FILES transmission-control_en_CA.ts)

set(PROJECT_SOURCES
        main.cpp
        mainwindow.cpp
        mainwindow.h
        mainwindow.ui
        rpc_client.h
        rpc_client.cpp
        torrent.h
        torrent.cpp
        dialogabout.cpp
        dialogabout.h
        dialogabout.ui
        version.h
        ${TS_FILES}
)

# The MACOSX_BUNDLE_ICON_FILE variable is added to the Info.plist
# generated by CMake. This variable contains the .icns file name,
# without the path.
set(MACOSX_BUNDLE_ICON_FILE planetary.icns)

# And the following tells CMake where to find and install the file itself.
set(app_icon_macos "${CMAKE_CURRENT_SOURCE_DIR}/Resources/icons/planetary.icns")
set_source_files_properties(${app_icon_macos} PROPERTIES
    MACOSX_PACKAGE_LOCATION "Resources")

if(${QT_VERSION_MAJOR} GREATER_EQUAL 6)
    qt_add_executable(Planetary
        MANUAL_FINALIZATION
        ${PROJECT_SOURCES}
        Resources/resources.qrc
        ${app_icon_macos}
        appsettings.ui
        appsettings.h
        appsettings.h
        appsettings.cpp
        Resources/icons/planetary.png
        torrentsortproxymodel.h torrentsortproxymodel.cpp
        serverconfig.h serverconfig.cpp serverconfig.ui
        percentfilldelegate.h percentfilldelegate.cpp
        torrentmodel.h torrentmodel.cpp
        Resources/icons/planetary_menu.png
        geoipservice.h geoipservice.cpp
        Resources/geoip/country.mmdb
    )

find_library(MAXMINDDB_LIBRARY maxminddb)
find_path(MAXMINDDB_INCLUDE_DIR maxminddb.h)

if(NOT MAXMINDDB_LIBRARY OR NOT MAXMINDDB_INCLUDE_DIR)
    message(WARNING "libmaxminddb not found; GeoIP MMDB lookup will not build")
else()
    message(STATUS "Found libmaxminddb: ${MAXMINDDB_LIBRARY}")
    message(STATUS "Found maxminddb.h: ${MAXMINDDB_INCLUDE_DIR}")
endif()

find_program(IWYU_PATH NAMES include-what-you-use iwyu)

#if(IWYU_PATH)
#    set_property(TARGET Planetary PROPERTY
#        CXX_INCLUDE_WHAT_YOU_USE
#        "${IWYU_PATH};-Xiwyu;--no_comments"
#    )
#endif()

# Define target properties for Android with Qt 6 as:
#    set_property(TARGET transmission-control APPEND PROPERTY QT_ANDROID_PACKAGE_SOURCE_DIR
#                 ${CMAKE_CURRENT_SOURCE_DIR}/android)
# For more information, see https://doc.qt.io/qt-6/qt-add-executable.html#target-creation

    qt_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})
else()
    if(ANDROID)
        add_library(Planetary SHARED
            ${PROJECT_SOURCES}
        )
# Define properties for Android with Qt 5 after find_package() calls as:
#    set(ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android")
    else()
        add_executable(Planetary
            ${PROJECT_SOURCES}
        )
    endif()

    qt5_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})
endif()

find_library(MAXMINDDB_LIBRARY maxminddb)
find_path(MAXMINDDB_INCLUDE_DIR maxminddb.h)

if(MAXMINDDB_LIBRARY AND MAXMINDDB_INCLUDE_DIR)
    message(STATUS "Found libmaxminddb: ${MAXMINDDB_LIBRARY}")
    message(STATUS "Found maxminddb.h: ${MAXMINDDB_INCLUDE_DIR}")

    target_compile_definitions(Planetary PRIVATE PLANETARY_HAVE_MAXMINDDB)
    target_include_directories(Planetary PRIVATE ${MAXMINDDB_INCLUDE_DIR})
    target_link_libraries(Planetary PRIVATE ${MAXMINDDB_LIBRARY})
else()
    message(WARNING "libmaxminddb not found; GeoIP MMDB lookup will be disabled")
endif()

set(GEOIP_DB_SOURCE
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources/geoip/country.mmdb"
)

if(EXISTS "${GEOIP_DB_SOURCE}")
    set_source_files_properties("${GEOIP_DB_SOURCE}" PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources/geoip"
    )

    target_sources(Planetary PRIVATE
        "${GEOIP_DB_SOURCE}"
    )
else()
    message(WARNING "GeoIP database not found at ${GEOIP_DB_SOURCE}")
endif()

set(APP_LICENSE_FILE
    "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE"
)

if(EXISTS "${APP_LICENSE_FILE}")
    set_source_files_properties("${APP_LICENSE_FILE}" PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources/licenses"
    )

    target_sources(Planetary PRIVATE
        "${APP_LICENSE_FILE}"
    )
else()
    message(WARNING "LICENSE file not found at ${APP_LICENSE_FILE}")
endif()

set(THIRD_PARTY_NOTICES_FILE
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources/licenses/THIRD_PARTY_NOTICES.txt"
)

if(EXISTS "${THIRD_PARTY_NOTICES_FILE}")
    set_source_files_properties("${THIRD_PARTY_NOTICES_FILE}" PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources/licenses"
    )

    target_sources(Planetary PRIVATE
        "${THIRD_PARTY_NOTICES_FILE}"
    )
endif()

target_link_libraries(Planetary PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt6::Network)

set_target_properties(Planetary PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER planetary.mvgrafx.net
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

install(TARGETS Planetary
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

if(QT_VERSION_MAJOR EQUAL 6)
    qt_finalize_executable(Planetary)
endif()
