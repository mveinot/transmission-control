cmake_minimum_required(VERSION 3.5)

project(Planetary VERSION 0.1 LANGUAGES CXX)

set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets LinguistTools Network Core Test)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets LinguistTools Network Core Test)

set(TS_FILES transmission-control_en_CA.ts)

set(PROJECT_SOURCES
        main.cpp
        mainwindow.cpp
        mainwindow.h
        mainwindow.ui
        rpc_client.h
        rpc_client.cpp
        torrent.h
        torrent.cpp
        dialogabout.cpp
        dialogabout.h
        dialogabout.ui
        ${TS_FILES}
)

# The MACOSX_BUNDLE_ICON_FILE variable is added to the Info.plist
# generated by CMake. This variable contains the .icns file name,
# without the path.
set(MACOSX_BUNDLE_ICON_FILE planetary.icns)

# And the following tells CMake where to find and install the file itself.
set(app_icon_macos "${CMAKE_CURRENT_SOURCE_DIR}/Resources/icons/planetary.icns")
set_source_files_properties(${app_icon_macos} PROPERTIES
    MACOSX_PACKAGE_LOCATION "Resources")

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/PLANETARY_VERSION" PLANETARY_VERSION_BASE)
string(STRIP "${PLANETARY_VERSION_BASE}" PLANETARY_VERSION_BASE)

if(NOT PLANETARY_VERSION_BASE MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "VERSION must be in major.minor.patch format, e.g. 0.8.0")
endif()

project(Planetary VERSION ${PLANETARY_VERSION_BASE} LANGUAGES CXX)

set(PLANETARY_VERSION_BUILD "0")

find_package(Git QUIET)

if(GIT_FOUND AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.git")
    execute_process(
        COMMAND "${GIT_EXECUTABLE}" rev-list --count HEAD
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        OUTPUT_VARIABLE PLANETARY_VERSION_BUILD
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET
    )
endif()

if(NOT PLANETARY_VERSION_BUILD MATCHES "^[0-9]+$")
    set(PLANETARY_VERSION_BUILD "0")
endif()

set(PLANETARY_VERSION_FULL
    "${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}.${PROJECT_VERSION_PATCH}.${PLANETARY_VERSION_BUILD}"
)

message(STATUS "Planetary version: ${PLANETARY_VERSION_FULL}")

configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/version.h.in"
    "${CMAKE_CURRENT_BINARY_DIR}/generated/version.h"
    @ONLY
)

configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
    "${CMAKE_CURRENT_BINARY_DIR}/Info.plist"
    @ONLY
)

#configure_file(
#    "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
#    "${CMAKE_CURRENT_BINARY_DIR}/Info.plist"
#    @ONLY
#)

file(WRITE
    "${CMAKE_CURRENT_BINARY_DIR}/planetary-version.txt"
    "${PLANETARY_VERSION_FULL}\n"
)

if(${QT_VERSION_MAJOR} GREATER_EQUAL 6)
    qt_add_executable(Planetary
        MANUAL_FINALIZATION
        ${PROJECT_SOURCES}
        Resources/resources.qrc
        ${app_icon_macos}
        appsettings.ui
        appsettings.h
        appsettings.cpp
        Resources/icons/planetary.png
        torrentsortproxymodel.h torrentsortproxymodel.cpp
        serverconfig.h serverconfig.cpp serverconfig.ui
        percentfilldelegate.h percentfilldelegate.cpp
        torrentmodel.h torrentmodel.cpp
        Resources/icons/planetary_menu.png
        geoipservice.h geoipservice.cpp
        Resources/geoip/country.mmdb
        singleinstanceguard.h singleinstanceguard.cpp
        settingskeys.h
        sessionsettingsdialog.h sessionsettingsdialog.cpp sessionsettingsdialog.ui
        torrentadddialog.h torrentadddialog.cpp torrentadddialog.ui
        torrentaddcontroller.h torrentaddcontroller.cpp
        watchfoldermanager.h watchfoldermanager.cpp
        updatechecker.h updatechecker.cpp
        Resources/licenses/gpl-2.0.txt
        Resources/translations/planetary_fr_CA.qm
        settingsimportexport.h settingsimportexport.cpp
        foldermapping.h
        foldermappingsdialog.h foldermappingsdialog.cpp foldermappingsdialog.ui
        bencodevalue.h bencodevalue.cpp
        bencodeparser.h bencodeparser.cpp
        torrentmetadataparser.h torrentmetadataparser.cpp
    )

target_include_directories(Planetary PRIVATE
    "${CMAKE_CURRENT_BINARY_DIR}/generated"
)

set_target_properties(Planetary PROPERTIES
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/Info.plist"
)

find_program(IWYU_PATH NAMES include-what-you-use iwyu)

# Define target properties for Android with Qt 6 as:
#    set_property(TARGET transmission-control APPEND PROPERTY QT_ANDROID_PACKAGE_SOURCE_DIR
#                 ${CMAKE_CURRENT_SOURCE_DIR}/android)
# For more information, see https://doc.qt.io/qt-6/qt-add-executable.html#target-creation

    qt_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})
else()
    if(ANDROID)
        add_library(Planetary SHARED
            ${PROJECT_SOURCES}
        )
# Define properties for Android with Qt 5 after find_package() calls as:
#    set(ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android")
    else()
        add_executable(Planetary
            ${PROJECT_SOURCES}
        )
    endif()

    qt5_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})
endif()

find_library(MAXMINDDB_LIBRARY maxminddb)
find_path(MAXMINDDB_INCLUDE_DIR maxminddb.h)

if(MAXMINDDB_LIBRARY AND MAXMINDDB_INCLUDE_DIR)
    message(STATUS "Found libmaxminddb: ${MAXMINDDB_LIBRARY}")
    message(STATUS "Found maxminddb.h: ${MAXMINDDB_INCLUDE_DIR}")

    target_compile_definitions(Planetary PRIVATE PLANETARY_HAVE_MAXMINDDB)
    target_include_directories(Planetary PRIVATE ${MAXMINDDB_INCLUDE_DIR})
    target_link_libraries(Planetary PRIVATE ${MAXMINDDB_LIBRARY})
else()
    message(WARNING "libmaxminddb not found; GeoIP MMDB lookup will be disabled")
endif()

set(GEOIP_DB_SOURCE
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources/geoip/country.mmdb"
)

if(EXISTS "${GEOIP_DB_SOURCE}")
    set_source_files_properties("${GEOIP_DB_SOURCE}" PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources/geoip"
    )

    target_sources(Planetary PRIVATE
        "${GEOIP_DB_SOURCE}"
    )
else()
    message(WARNING "GeoIP database not found at ${GEOIP_DB_SOURCE}")
endif()

set(APP_LICENSE_FILE
    "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE"
)

if(EXISTS "${APP_LICENSE_FILE}")
    set_source_files_properties("${APP_LICENSE_FILE}" PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources/licenses"
    )

    target_sources(Planetary PRIVATE
        "${APP_LICENSE_FILE}"
    )
else()
    message(WARNING "LICENSE file not found at ${APP_LICENSE_FILE}")
endif()

set(THIRD_PARTY_NOTICES_FILE
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources/licenses/THIRD_PARTY_NOTICES.txt"
)

if(EXISTS "${THIRD_PARTY_NOTICES_FILE}")
    set_source_files_properties("${THIRD_PARTY_NOTICES_FILE}" PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources/licenses"
    )

    target_sources(Planetary PRIVATE
        "${THIRD_PARTY_NOTICES_FILE}"
    )
endif()

#target_link_libraries(Planetary PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt6::Network)
target_link_libraries(Planetary PRIVATE
    Qt${QT_VERSION_MAJOR}::Widgets
    Qt${QT_VERSION_MAJOR}::Network
)

set_target_properties(Planetary PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER planetary.mvgrafx.net
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/Info.plist"
    WIN32_EXECUTABLE TRUE
)

include(GNUInstallDirs)

install(TARGETS Planetary
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    BUNDLE DESTINATION .
)

install(FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources/icons/planetary.png"
    DESTINATION "${CMAKE_INSTALL_DATAROOTDIR}/icons/hicolor/256x256/apps"
    RENAME planetary.png
)

install(FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/packaging/linux/planetary.desktop"
    DESTINATION "${CMAKE_INSTALL_DATAROOTDIR}/applications"
)

install(FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources/geoip/dbip-country-lite-2026-06.mmdb"
    DESTINATION "${CMAKE_INSTALL_DATAROOTDIR}/planetary/geoip"
    OPTIONAL
)

install(FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE"
    "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.txt"
    DESTINATION "${CMAKE_INSTALL_DATAROOTDIR}/planetary/licenses"
    OPTIONAL
)

install(TARGETS Planetary
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

enable_testing()
add_subdirectory(tests)

if(QT_VERSION_MAJOR EQUAL 6)
    qt_finalize_executable(Planetary)
endif()
