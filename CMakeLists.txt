foreach(PLANETARY_TS_FILE IN LISTS PLANETARY_TS_FILES)
    get_filename_component(PLANETARY_TRANSLATION_NAME "${PLANETARY_TS_FILE}" NAME_WE)

    set(PLANETARY_QM_FILE
        "${PLANETARY_GENERATED_TRANSLATIONS_DIR}/${PLANETARY_TRANSLATION_NAME}.qm"
    )

    add_custom_command(
        OUTPUT "${PLANETARY_QM_FILE}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${PLANETARY_GENERATED_TRANSLATIONS_DIR}"
        COMMAND Qt${QT_VERSION_MAJOR}::lrelease "${PLANETARY_TS_FILE}" -qm "${PLANETARY_QM_FILE}"
        DEPENDS "${PLANETARY_TS_FILE}"
        COMMENT "Generating ${PLANETARY_TRANSLATION_NAME}.qm"
        VERBATIM
    )

    list(APPEND PLANETARY_QM_FILES "${PLANETARY_QM_FILE}")
    string(APPEND PLANETARY_TRANSLATION_QRC_ENTRIES
        "        <file alias=\"translations/${PLANETARY_TRANSLATION_NAME}.qm\">translations/${PLANETARY_TRANSLATION_NAME}.qm</file>\n"
    )
endforeach()

# Drive rcc explicitly so its generated C++ output has a direct file-level
# dependency on every lrelease output. AutoRCC's generated helper target can
# otherwise race lrelease after a fresh configure in parallel builds.
add_custom_command(
    OUTPUT "${PLANETARY_GENERATED_TRANSLATIONS_CPP}"
    COMMAND Qt${QT_VERSION_MAJOR}::rcc
            --name planetary_translations
            --output "${PLANETARY_GENERATED_TRANSLATIONS_CPP}"
            "${PLANETARY_GENERATED_TRANSLATIONS_QRC}"
    DEPENDS
        "${PLANETARY_GENERATED_TRANSLATIONS_QRC}"
        ${PLANETARY_QM_FILES}
    COMMENT "Compiling Planetary translations"
    VERBATIM
)

set_source_files_properties(
    "${PLANETARY_GENERATED_TRANSLATIONS_CPP}"
    PROPERTIES
        GENERATED TRUE
        SKIP_AUTOGEN TRUE
)

if(WIN32)

    find_path(MAXMINDDB_INCLUDE_DIR
        NAMES maxminddb.h
        PATHS "${MAXMINDDB_ROOT}/include"

    )

    find_library(MAXMINDDB_LIBRARY
        NAMES maxminddb libmaxminddb
        PATHS "${MAXMINDDB_ROOT}/lib"
        NO_DEFAULT_PATH
    )

    message(STATUS "MAXMINDDB_INCLUDE_DIR=${MAXMINDDB_INCLUDE_DIR}")
    message(STATUS "MAXMINDDB_LIBRARY=${MAXMINDDB_LIBRARY}")

    if(MAXMINDDB_INCLUDE_DIR AND MAXMINDDB_LIBRARY)
        target_include_directories(Planetary PRIVATE
            "${MAXMINDDB_INCLUDE_DIR}"
        )

        target_link_libraries(Planetary PRIVATE
            "${MAXMINDDB_LIBRARY}"
        )

        target_compile_definitions(Planetary PRIVATE
            PLANETARY_WITH_GEOIP=1
        )

        message(STATUS "GeoIP support enabled on Windows")
    else()
        message(WARNING "libmaxminddb not found on Windows; lookup will be disabled")
    endif()
endif()
